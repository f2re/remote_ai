cmake_minimum_required(VERSION 3.10)
project(remote-ai-astra VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

find_package(Qt5 5.11 REQUIRED COMPONENTS Core Gui Widgets Test)

add_executable(remote-ai-astra src/main.cpp resources/mascot.qrc)
target_include_directories(remote-ai-astra PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(remote-ai-astra PRIVATE Qt5::Widgets)
set_target_properties(remote-ai-astra PROPERTIES AUTOMOC ON AUTORCC ON)

install(TARGETS remote-ai-astra RUNTIME DESTINATION bin)
install(FILES packaging/remote-ai-astra.desktop DESTINATION share/applications)
install(FILES ../Sources/RemoteAI/Resources/Mascot/mascot-waiting.png
  DESTINATION share/icons/hicolor/128x128/apps RENAME remote-ai-astra.png)

include(CTest)
if(BUILD_TESTING)
  add_executable(remote-ai-astra-tests tests/core_tests.cpp)
  target_include_directories(remote-ai-astra-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(remote-ai-astra-tests PRIVATE Qt5::Test Qt5::Core)
  add_test(NAME core-contract COMMAND remote-ai-astra-tests)
  add_test(NAME offscreen-smoke COMMAND remote-ai-astra --smoke)
  set_tests_properties(offscreen-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

set(ASTRA_TARGET "debian12" CACHE STRING "Compatibility baseline: debian10, debian12, astra17, astra18")
set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGE_NAME "remote-ai-astra")
set(CPACK_PACKAGE_VENDOR "Remote AI")
set(CPACK_PACKAGE_CONTACT "https://github.com/f2re/remote_ai/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Русскоязычный маскот для удалённого Claude Code на Linux")
set(CPACK_PACKAGE_VERSION "0.3.0")
set(CPACK_DEBIAN_PACKAGE_VERSION "0.3.0~rc1")
set(CPACK_DEBIAN_PACKAGE_RELEASE "1+${ASTRA_TARGET}")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "amd64")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
include(CPack)
